% Beamer color scheme — replicates defense.pdf deep blue style
% Author, Title, Date, Page X/N in footline; blue top bar; blue bullets

\definecolor{navyblue}{HTML}{003366}
\definecolor{lightgray}{HTML}{F0F0F0}
\definecolor{darkred}{HTML}{8B0000}
\definecolor{darkgreen}{HTML}{005030}

% --- Structure colors (top bar, footline, bullets, navigation) ---
\setbeamercolor{structure}{fg=navyblue,bg=navyblue}
\setbeamercolor{palette primary}{fg=white,bg=navyblue}
\setbeamercolor{palette secondary}{fg=white,bg=navyblue!80}
\setbeamercolor{palette tertiary}{fg=white,bg=navyblue!60}
\setbeamercolor{palette quaternary}{fg=navyblue,bg=white}

% --- Title page ---
\setbeamercolor{title}{fg=white}
\setbeamercolor{subtitle}{fg=white!90}
\setbeamercolor{titlelike}{fg=white}
\setbeamercolor{author}{fg=black}
\setbeamercolor{date}{fg=black!70}

% --- Frame title (top bar) ---
\setbeamercolor{frametitle}{fg=white,bg=navyblue}
\setbeamercolor{frametitle right}{fg=white,bg=navyblue!80}

% --- Footline ---
\setbeamercolor{footline}{fg=white,bg=navyblue}
\setbeamercolor{page number in head/foot}{fg=white,bg=navyblue}

% --- Blocks ---
\setbeamercolor{block title}{fg=white,bg=navyblue}
\setbeamercolor{block body}{fg=black,bg=lightgray}

\setbeamercolor{alertblock title}{fg=white,bg=darkred}
\setbeamercolor{alertblock body}{fg=black,bg=lightgray}

\setbeamercolor{exampleblock title}{fg=white,bg=darkgreen}
\setbeamercolor{exampleblock body}{fg=black,bg=lightgray}

% --- Items ---
\setbeamercolor{item}{fg=navyblue}
\setbeamercolor{subitem}{fg=navyblue!80}
\setbeamercolor{item projected}{fg=white,bg=navyblue}

% --- TOC ---
\setbeamercolor{section in toc}{fg=navyblue}
\setbeamercolor{subsection in toc}{fg=navyblue!80}

% --- Caption ---
\setbeamercolor{caption name}{fg=navyblue}
